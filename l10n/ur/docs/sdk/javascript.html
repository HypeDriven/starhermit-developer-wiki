JavaScript SDK (براؤزر گیمز)

starhermit-sdk.js براؤزر گیمز کے لیے ایک واحد فائل، انحصار سے پاک کلائنٹ ہے۔ StarHermit پر شائع ہوا۔ یہ لانچ ٹوکن پڑھتا ہے، اس کی تجدید کرتا رہتا ہے، اور عام لوگوں کے لیے مددگار فراہم کرتا ہے۔ گیم اسکوپڈ لانچ ٹوکن APIs: شناخت، کلاؤڈ سیو، فی کھلاڑی کی ترتیبات اور کنٹرولز، دعوتیں، میچ میکنگ، سیشنز اور گیم پلے ساکٹ، کامیابیاں، لیڈر بورڈ، ری پلے، چیٹ، آواز اور حقیقی وقت کے کمرے۔
فائل docs/sdk/starhermit-sdk.js - اسے بغیر کسی تبدیلی کے اپنے گیم میں کاپی کریں۔
عالمی window.StarHermit (بھی module.exports نوڈ کے تحت، ٹیسٹ کے لیے)
انحصار کوئی نہیں - یہ براؤزر کا اپنا استعمال کرتا ہے۔ fetch, WebSocket اور (کمپریسڈ سیو پڑھنے کے لیے) DecompressionStream
API بیس لائن REST v1 اور WebSocket v1، صرف لانچ ٹوکن سطح

SDK ایک کلائنٹ ہے، پلیٹ فارم کا دوسرا نفاذ نہیں۔ اسکور اور کامیابی کھل جاتی ہے۔ صرف گیم کے سرور کے ذریعہ لکھا جاتا ہے (اسکرپٹ کا نتیجہ یا کنٹینر کنٹرول پیغام) - ایک کلائنٹ انہیں کبھی بھی پوسٹ نہیں کر سکتا، اور SDK انہیں صرف پڑھتا ہے۔ باقی سب کچھ سرور سے مستند رہتا ہے۔ بالکل اسی طرح جیسے پر دستاویزی ہے۔ API صفحات.

تصدیق شدہ مددگار لانچ ٹوکن کے بغیر کوئی API درخواستیں نہیں کرتے ہیں۔ پڑھتا ہے عام طور پر حل کو null, []، یا ایک خالی نقشہ؛ لکھنے سے حل ہوسکتا ہے۔ null/false اور ڈیباؤنس شدہ بچتوں کا کوئی فوری نہیں ہے۔ واپسی کی قیمت ایک ساکٹ منقطع رہتا ہے۔ اگر کھلاڑی نے انتخاب کیا ہے تو اسٹینڈ اسٹون پیشرفت کو میموری میں رکھیں غیر ٹریک شدہ کھیل کال کریں۔ signOut() اکاؤنٹ پلے سے اس موڈ میں سوئچ کرنے سے پہلے تاکہ تجدید رک جائے۔

انسٹال کریں۔

فائل کو اپنے گیم کے دیگر جامد اسکرپٹس کے ساتھ کاپی کریں — جہاں بھی آپ کی بلڈ انہیں بھیجتی ہے — اور لوڈ کریں یہ آپ کے اپنے کوڈ سے پہلے:

<script src="starhermit-sdk.js"></script>
<script>
  StarHermit.init();          // as early as possible: it reads and strips the token from the URL fragment
</script>

کال کریں۔ init() ایک بار، کسی بھی راؤٹر کو چھونے سے پہلے location.hash. اختیارات: init({ base, gameId, autoRefresh, launcherUrl }) — base پوائنٹس REST کالز ایک مجاز اصل پر مقامی ترقی کے دوران (پلیٹ فارم کی میزبانی والے گیمز ایک ہی اصل کا استعمال کرتے ہیں)۔ gameIdاگر فراہم کیا جائے تو مماثل ہونا چاہیے۔ سائن ان کے لیے میزبان UUID؛ مقامی/بیرونی صفحات درست گیم سائن ان کی واپسی کی اصل نہیں ہیں۔ autoRefresh: false ٹوکن کی تجدید کو ایک گیم میں چھوڑ دیتا ہے۔ اپنی زنجیر چلاتا ہے، اور launcherUrl جہاں اوور رائیڈ کرتا ہے۔ relaunch() پلیئر بھیجتا ہے (default https://dashboard.starhermit.com/).

سائن ان اور شناخت

ضرورت کال کریں۔
کیا کوئی کھلاڑی سائن ان ہے؟ StarHermit.signedIn, StarHermit.userId, StarHermit.slug
سائن ان / سائن آؤٹ / ختم ہونے پر رد عمل ظاہر کریں۔ StarHermit.on('auth', ({ signedIn, reason }) => …)
دکھائیں a سائن ان کریں۔ بٹن صرف جب StarHermit.canSignIn() (ٹوکن کے بغیر گیم UUID ذیلی ڈومین پر درست)؛ کلک پر StarHermit.signIn()
کھلاڑی کا نام / اوتار await StarHermit.profile() → { userId, username, nickname, displayName }; StarHermit.avatarUrl()
دوستو await StarHermit.friends()

SDK لانچ کی دونوں شکلیں ہینڈل کرتا ہے۔ #game_token=<jwt>[&session_id=…] لائبریری سے یا ایک دعوت دیں، اور #access_token=<jwt> براہ راست سائن ان سے واپس آنا — اور ٹوکن کے ذریعے تجدید کرتا ہے۔ POST /api/v1/games/{slug}/launch-token اس کی میعاد ختم ہونے سے پہلے (دیکھیں۔ ٹوکن لانچ کریں)۔ ڈسپلے کا نام پروفائل ہے۔ nickname; /api/v1/me نہیں ہے لانچ ٹوکن کے ساتھ قابل رسائی۔

کلاؤڈ سیوز

کھیل کی سلاٹ ہے game:<slug> (catalog: کھیل کے اندر سے بچت); SDK راستہ بناتا ہے، پے لوڈ کو زپ کرتا ہے اور ہینڈل کرتا ہے۔ /info چیک کریں

const remote = await StarHermit.loadJSON();          // null: empty slot, signed out — or a failed read
if (StarHermit.saveLoadFailed()) showSyncError();    // tell those apart when it matters
else if (remote && isNewer(remote, local)) adopt(remote);

StarHermit.saveJSON(state);                          // debounced (2 s) checkpoint upload
addEventListener('pagehide', () => StarHermit.flushSave(true));   // keepalive flush
StarHermit.on('saved', ok => setSyncIndicator(ok));

محفوظ کرنے سے پہلے لوڈ کریں۔ کسی بھی تحریر تک پہنچنے سے پہلے کلاؤڈ کاپی کا موازنہ کریں اور اسے اپنا لیں۔ saveJSON - بشمول ترتیبات، عرفی نام یا چیک پوائنٹ بوٹ پر لکھتا ہے - یا باسی مقامی کاپی کر سکتے ہیں۔ کسی دوسرے آلے پر بنائے گئے نئے سیو کو اوور رائٹ کریں۔ ابتدائی بچت کو اس وقت تک روکیں جب تک کہ اسٹارٹ اپ لوڈ ختم نہ ہوجائے۔

SDK اس ریس کے دوسرے نصف حصے کی حفاظت کرتا ہے: جب کوئی پڑھنا ناکام ہوجاتا ہے (نیٹ ورک یا سرور کی خرابی، جیسا کہ ایک خالی سلاٹ کے خلاف) یہ سیٹ کرتا ہے۔ saveLoadFailed()، خارج کرتا ہے۔ saveerror، اور انکار جب تک لکھتا ہے۔ /info اس بات کی تصدیق کرتا ہے کہ سلاٹ خالی ہے یا بعد میں loadSave() کامیاب ہوتا ہے ایک انکار شدہ تحریر حل ہو جاتی ہے۔ false اور خارج کرتا ہے saved کے ساتھ false.

بوجھ کے بعد لکھنا مشروط ہے۔ SDK کو ورژن یاد ہے (ETag) اسے آخری محفوظ کریں پڑھیں یا لکھیں اور اسے واپس بھیج دیں، لہذا ایک تحریر صرف اس محفوظ پر اترتی ہے جس پر یہ مبنی تھا — یا، ایک سلاٹ کے لیے وہ خالی تھا، صرف اس وقت تک جب یہ اب بھی ہے۔ جب درمیان میں کوئی دوسرا آلہ محفوظ ہو جاتا ہے تو لکھنے سے انکار کر دیا جاتا ہے۔ (412) SDK خارج کرتا ہے۔ saveconflict، سیٹ saveConflicted()، اور انکار مزید لکھتا ہے جب تک کہ a loadSave() گیم کو ضم کرنے یا نئے سیو کو اپنانے دیتا ہے۔ ایک کھیل جو کبھی لوڈ نہیں ہوتا لکھتا ہے۔ غیر مشروط طور پر، پہلے کی طرح.

StarHermit.on('saveconflict', async () => {
  const remote = await StarHermit.loadJSON();       // lifts the block
  StarHermit.saveJSON(merge(remote, local));        // conditional on the version just loaded
});

نچلی سطح: loadSave() / writeSave(text) خام تاروں کے لیے، saveInfo() کے لیے { exists, sizeBytes, updatedAt, etag }.

کریش اور بگ رپورٹس

کھلاڑی کھیل کے اندر سے مسائل کی اطلاع دے سکتے ہیں۔ مالک ان کے ذریعے ٹرائی کرتا ہے۔ رپورٹس API.

await StarHermit.reportBug('Score did not save', 'Finished level 3, score shows 0', {
  clientVersion: '1.4.2',
  attachments: [{ fileName: 'state.json', contentType: 'application/json', data: JSON.stringify(state) }],
});
StarHermit.captureCrashes({ clientVersion: '1.4.2' });   // opt-in: uncaught errors become crash reports
const mine = await StarHermit.myReports();               // [{ id, kind, status, title, ... }]

report(kind, title, description, opts) عام شکل ہے (reportBug اور reportCrash ٹھیک کریں قسم)۔ captureCrashes فائلیں زیادہ سے زیادہ ایک رپورٹ فی الگ پیغام اور تین فی صفحہ لوڈ، لہذا a کریش لوپ کھلاڑی کے یومیہ الاؤنس (20 فی گیم بطور ڈیفالٹ) استعمال نہیں کر سکتا۔ اس سے پہلے کھلاڑی سے پوچھیں۔ کسی بھی ذاتی چیز کو منسلک کرنا۔

ترتیبات اور کنٹرولز

ضرورت کال کریں۔
فی کھلاڑی کی ترجیحات (حجم، گرافکس پیش سیٹ، زبان، رسائی…) getSettings(), getSetting(key), setSetting(key, value), patchSettings({…}) (a null قدر کلید کو ہٹا دیتا ہے) deleteSetting(key), clearSettings()
کی بورڈ بائنڈنگز کا اعلان کر دیا گیا۔ starhermit.txt کے طور پر control.<action>=… await loadBindings(defaults) → { action: codes[] } پلیئر کے اوور رائیڈز لاگو ہونے کے ساتھ
ایک ری بائنڈنگ اسکرین getControls(), setControl(action, codes), setControls(bindings), resetControls()

راستہ keydown کے ذریعے event.code اور سے پابندیاں loadBindings. دیکھیں اپنے گیم میں کنٹرولز شامل کریں۔ اور starhermit.txt ظاہر.

گرافکس مینو کے لیے، graphics-options.js سیٹنگز کالز کے اوپر بیٹھتا ہے: یہ معیار کے پیش سیٹ کا خود بخود پتہ لگاتا ہے، کھلاڑی کی پسند کو برقرار رکھتا ہے۔ localStorage اور ان کا فی گیم ترتیبات، اور ایک ریڈی میڈ پینل بھیجتا ہے۔ دیکھیں خود کار طریقے سے پتہ چلنے والے معیار کے ساتھ گرافکس کے اختیارات.

آڈیو سیکشن کے لیے، دیکھیں آڈیو کے اختیارات جس میں خود بخود پتہ چلنے والی آواز کی حد ہے۔: تجویز کردہ حجم اور آواز کے کنٹرول، خالص audio-options.js مددگار، اور آپ کے مکسر کے ذریعہ نافذ کردہ پلے بیک کیپ۔

زبان کے حصے کے لیے، دیکھیں اپنے کھیل کو مقامی بنائیں اور زبان کی ترتیبات شامل کریں۔: دس زبانوں کے ترجمے کے پیک، SVG جھنڈے، دائیں سے بائیں لے آؤٹ، اور ایک ui.language ترجیح ذخیرہ گیم کی ترتیبات کے ساتھ۔ استعمال کریں۔ api() جب آپ کو سیٹنگز کو سرفنگ کرنے کی ضرورت ہوتی ہے تو پڑھنے/لکھنے کی ناکامیاں۔

ملٹی پلیئر، دعوتیں اور سماجی

سیشن گیم پلے، میچ میکنگ، گیم کی کامیابیوں، لیڈر بورڈز، اور ری پلے کو سرور کی ضرورت ہے۔ (سکرپٹ یا کنٹینر)۔ ریئل ٹائم کمرے اور دوست نہیں کرتے. سیشن چیٹ اور آواز کو متعلقہ گفتگو کی رکنیت کی ضرورت ہے۔

علاقہ کالز
سیشنز mySessions(), getSession(id), sessionRecovery(id), leaveSession(id), startAiSession(); StarHermit.launchSessionId دعوت کے آغاز کے بعد
گیم پلے ساکٹ const conn = connect(sessionId, { onGame, onError, onPresence, onAchievement, onResumed, onAbandoned, onOpen, onClose, onAuthLost })، پھر conn.send(data); بیک آف کے ساتھ دوبارہ جڑتا ہے، پہلے ٹوکن کی تجدید کرتا ہے، اور نارمل/پالیسی/اوور سائز بند ہونے پر رک جاتا ہے (1000/1008/1009)، 4403/4404، یا جب تجدید سے انکار کر دیا جاتا ہے (دیکھیں دوبارہ جوڑ رہا ہے)
میچ میکنگ queues(), joinQueue([keys]), matchStatus(), cancelMatch(), waitForMatch(opts)
دعوت دیتا ہے۔ sendInvite(toUserId[, sessionId]) (ایک کے ساتھ sessionId دوست اس رننگ سیشن میں شامل ہوتا ہے) invites(), acceptInvite(id), declineInvite(id); inviteLink(query) شیئر لنک کے لیے
اپنے لیڈر بورڈز پر رن ​​پوسٹ کریں۔ submitScores({ boardKey: number }) → قبول شدہ چابیاں (دیکھیں۔ سنگل پلیئر لیڈر بورڈز)
کامیابیاں اور لیڈر بورڈ (صرف پڑھنے کے لیے) achievements(), linkedAchievements(otherSlug), leaderboards(), leaderboard(key, opts), leaderboardEntries(boardId, opts); کے ذریعے ٹوسٹس کو غیر مقفل کریں۔ on('achievement', …)
ری پلے کرتا ہے۔ myReplays(limit), getReplay(id)
سیشن چیٹ chatMessages(id), sendChat(id, text), pollChat(id, onMessages, intervalMs) - صرف لانچ ٹوکن کے لیے آرام اور پولنگ
وائس / ریئل ٹائم کمرے StarHermit.voice.* (باقی؛ WebRTC سگنلنگ گیم پر چھوڑ دی گئی ہے) StarHermit.realtime.* (کمرے پلس ws/v1/realtime)

کوئی بھی چیز لپیٹ نہیں ہے ایک کال کی دوری پر ہے: StarHermit.api(path, { method, body }) ایک بھیجتا ہے تصدیق شدہ JSON درخواست (حل کرتی ہے۔ null 404/204 پر یا سائن آؤٹ ہونے پر، مسترد کر دیتا ہے۔ { status, message } دوسری ناکامیوں پر، 401 پر ایک بار تجدید)۔

سنگل پلیئر لیڈر بورڈز

ایک کلائنٹ کبھی بھی اسکور پوسٹ نہیں کر سکتا — صرف گیم کا سرور کر سکتا ہے — اس لیے سنگل پلیئر گیم کی ضرورت ہوتی ہے۔ اپنے بورڈز کے مالک ہونے کے لیے چھوٹا پلیٹ فارم اسکرپٹ۔ score-script.js ایک ہے: اسے اگلا کاپی کریں۔ اپنے کھیل میں، اس کا اعلان کریں۔ starhermit.txt کے طور پر server=score-script.js، اور صرف اس میں ترمیم کریں۔ BOARDS بلاک، جو ہر بورڈ کو آپ کے طور پر درج کرتا ہے۔ اسے بنائیں:

var BOARDS = {
  "high-score": { "name": "High score", "scoreType": "integer", "sortDirection": "desc", "minScore": 0, "maxScore": 1000000 }
};

جب رن ختم ہو جائے تو اسے پوسٹ کریں اور کھلاڑی کی جگہ دکھائیں:

const accepted = await StarHermit.submitScores({ 'high-score': total });   // [] when signed out or refused
if (accepted.includes('high-score')) {
  const board = await StarHermit.leaderboard('high-score', { pageSize: 100 });
  const me = board.items.find(e => e.userId === StarHermit.userId);
  showRank(me && me.rank);
}

submitScores ایک پریکٹس سیشن کھولتا ہے (POST .../sessions/ai)، بھیجتا ہے۔ {type:'result', scores} پر گیم پلے ساکٹ، اور اسکرپٹ کی قبول کردہ بورڈ کیز کو حل کرتا ہے۔ یہ بھی خارج کرتا ہے scores کے ساتھ وہ فہرست. اسکرپٹ ایک اسکور کو گراتا ہے جو بورڈ کی قسم کا نمبر نہیں ہے یا اس سے باہر آتا ہے۔ minScore/maxScore، باقی اس کے ذریعے پوسٹ کرتا ہے۔ scores واپسی، اور سیشن ختم. کلائنٹ اب بھی نتیجہ کی اطلاع دیتا ہے. رینج میں ایجاد کردہ سکور گزر سکتا ہے۔ یہ بورڈ آرام دہ اور پرسکون کے لئے موزوں ہیں کھیلیں مقابلے کے لیے، انعام دینے سے پہلے اسکرپٹ یا کنٹینر میں رن کی توثیق کریں۔ اعتراف اسکرپٹ سے منظور شدہ کلیدوں کی فہرست۔ یہ اس بات کا ثبوت نہیں ہے کہ ہر بورڈ فعال تھا یا ایک نیا ذاتی بہترین ذخیرہ کیا گیا تھا. دیکھیں اعتماد اور دستیابی.

دوبارہ منسلک اور میعاد ختم ہونے والے ٹوکن

ایک ساکٹ اپنے URL میں لانچ ٹوکن رکھتا ہے، اور اس سے پہلے ختم شدہ ٹوکن کے ساتھ مصافحہ کرنے سے انکار کر دیا جاتا ہے۔ اپ گریڈ - براؤزر صرف قریبی کوڈ کی اطلاع دیتا ہے۔ 1006بالکل نیٹ ورک ڈراپ کی طرح۔ تو ایک ناکام دوبارہ جڑنا ایک توثیق کی ناکامی ہو سکتی ہے، اور اسی URL کو دوبارہ کوشش کرنے سے کبھی بھی بازیافت نہیں ہو سکتی۔ تجدید صرف کام کرتی ہے۔ جبکہ موجودہ ٹوکن درست ہے اور اصل لانچ کے 12 گھنٹے کے اندر؛ اس کے بعد صرف لانچر ایک نیا ٹوکن لگا سکتا ہے۔

connect() اسے خود ہینڈل کرتا ہے: ہر دوبارہ جڑنے سے پہلے یہ ٹوکن کی تجدید کرتا ہے۔ POST /api/v1/games/{slug}/launch-token (جب تک کہ ساکٹ کھلنے کے بعد سے اس کی تجدید نہ ہو) اور دوبارہ کھل جائے۔ نئے کے ساتھ. تجدید کے دوران نیٹ ورک یا سرور کی خرابی پیچھے ہٹ جاتی ہے اور دوبارہ تجدید ہوتی ہے - ایسا کبھی نہیں ہوتا ہے۔ پرانے URL کو دوبارہ کھولتا ہے۔ جب تجدید سے انکار کر دیا جاتا ہے یا ٹوکن کی میعاد ختم ہو جاتی ہے، تو SDK سائن آؤٹ ہو جاتا ہے (auth کے ساتھ reason: 'expired')، دوبارہ جڑنا اور کال کرنا بند کر دیتا ہے۔ onAuthLost():

const conn = StarHermit.connect(sessionId, {
  onGame, onClose,
  onAuthLost: () => showReconnectPanel(),   // e.g. "Session expired — Back to StarHermit"
});
backButton.onclick = () => StarHermit.relaunch();

relaunch() پلیئر کو ایک تازہ لانچ ٹوکن کے لیے واپس بھیجتا ہے: گیم کھولنے پر سائن ان کے ذریعے اس طرح (StarHermit.launchKind === 'sign-in')، بصورت دیگر لانچر کو۔ لانچر کے اندر گیم فریم یہ سب سے اوپر والی ونڈو کو نیویگیٹ کرتا ہے، جسے براؤزر صرف صارف کے اشارے سے اجازت دیتے ہیں، لہذا اسے کال کریں۔ ایک کلک سے؛ یہ واپس آتا ہے false جب نیویگیشن سے انکار کر دیا گیا تھا۔

ایک براؤزر ختم سیشن، شرائط، رکنیت، یا ناکامیوں کو اس طرح چھپا سکتا ہے۔ 1006. ایک سے پہلے خودکار دوبارہ کوشش، connect() REST کے ذریعے سیشن کی حالت چیک کرتا ہے اور، کب opts.build فراہم کیا جاتا ہے، گیم کی معلومات سے اس کا موازنہ کرتا ہے۔ onUnavailable(error) مستقل REST کی ناکامی یا ختم ہو گئی۔ سیشن onTermsRequired(body) واضح شرائط کی قبولیت کی درخواست کرتا ہے۔ اس کے بعد نیا کنکشن بنائیں ان کو حل کرنا. onClose(1008) اس کا مطلب ہے کہ کنکشن ختم یا پالیسی کی خلاف ورزی کی گئی تھی۔ نہ کرو خود بخود اسے دوبارہ کھولیں.

ساکٹ آپ کا گیم خود کھلتا ہے (realtime.socketUrl(roomId), voice.socketUrl(roomId)) کی ضرورت ہے ایک ہی اصول ایک کو دوبارہ کھولنے سے پہلے، کال کریں۔ renewForReconnect() اور اس کے نتیجے پر عمل کریں:

async function reopen() {
  const r = await StarHermit.renewForReconnect();
  if (r === 'renewed') ws = new WebSocket(StarHermit.realtime.socketUrl(roomId)); // fresh token in the URL
  else if (r === 'retry') setTimeout(reopen, backoff());                         // renewal itself failed
  else if (r === 'terms-required') showTermsPanel();                              // explicit acceptance
  else showReconnectPanel();                                                     // 'relaunch'
}

ہم وقتی تجدید ایک درخواست کا اشتراک کرتے ہیں۔ terms-required پلیئر ٹوکن کو محفوظ رکھتا ہے اور خارج کرتا ہے۔ termsrequired; ڈسپلے getTerms() اور کال کریں acceptTerms(displayed.hash) صرف کھلاڑی پر کارروائی قبول کریں، پھر دوبارہ کوشش کریں۔ آپ کے گیم کا انتظام کرنے والے ساکٹ کو بھی ختم شدہ وسائل کے لیے REST چیک کی ضرورت ہوتی ہے۔ اور پالیسی بند ہونے پر رکنا چاہیے۔ دیکھیں خرابیوں کا سراغ لگانا.
