ترقی کو محفوظ طریقے سے لوڈ اور محفوظ کریں۔

سائن ان کردہ کھلاڑی کے کلائنٹ کی ملکیت میں پیش رفت کے لیے کلاؤڈ سیو کا استعمال کریں۔ غیر دستخط شدہ واحد کھلاڑی کا اختیار یادداشت میں پیشرفت کو برقرار رکھتا ہے اور اکاؤنٹ کو پڑھتا یا لکھتا نہیں بچاتا ہے۔ ترجیحات کا تعلق ہے۔ کھیل کی ترتیبات، ترقی کی دستاویز نہیں۔

لانچ ٹوکن بالکل رسائی حاصل کر سکتا ہے۔ game:<gameId>، کہاں gameId اس کا ہے game_scope. پلیٹ فارم مبہم بائٹس ذخیرہ کرتا ہے، فی سلاٹ ایک بچت کے ساتھ۔ آپ کے گیم کی توقع کے فارمیٹ کی توثیق کریں۔ دیکھیں کلاؤڈ سیوز کوٹہ اور مشروط ہیڈرز کے لیے۔

ایک چھوٹا مکمل JSON نسخہ

کاپی cloud-save-store.js آپ کی تعمیر میں. یہ مثال خام ذخیرہ کرتی ہے۔ UTF-8 JSON بائٹس؛ پلیٹ فارم ان کی خدمت کرتا ہے۔ application/zip لیکن زپ فارمیٹ کی ضرورت نہیں ہے۔ زپ سیو کے لیے ایک مختلف ریڈر استعمال کریں، بشمول براؤزر SDK کا زپ محفوظ کرتا ہے۔. موجودہ کھلاڑیوں کے لیے منتقلی کے منصوبے کے بغیر فارمیٹس کو تبدیل نہ کریں۔

import { CloudSaveStore } from './cloud-save-store.js';

const store = new CloudSaveStore({
  gameId: StarHermit.slug,
  token: () => StarHermit.token, // Always obtain the current renewed token.
});
const valid = value => value?.version === 1 && Number.isSafeInteger(value.points) && value.points >= 0;
let progress = { version: 1, points: 0 };
let saving = false;

async function startAccountGame() {
  // Keep gameplay/saving disabled until this resolves.
  const remote = await store.load(valid);
  progress = remote ?? { version: 1, points: 0 };
  render(progress);
}

async function saveCheckpoint() {
  if (saving) return; // Coalesce checkpoints; do not issue overlapping writes.
  saving = true;
  showSync('Saving…');
  try {
    await store.save(progress);
    showSync('Saved');
  } catch (error) {
    if (error.status === 412) {
      // Keep local progress in memory. Offer Reload cloud or an explicit merge.
      showConflict(() => startAccountGame());
    } else showSync('Save failed; retry after resolving the error');
  } finally {
    saving = false;
  }
}

render, showSync، اور showConflict آپ کے UI ہیں۔ انتظار کرو startAccountGame() چالو کرنے سے پہلے کھیلنا یا بچانا۔ لوڈ کی ناکامی پر، کھلاڑی کی موجودہ مقامی حالت کو برقرار رکھیں، ناکامی کو ظاہر کریں، اور اپ لوڈز کو بلاک رکھیں۔ اے 404 خالی سلاٹ کیس ہے؛ 401, 403, 5xx, خراب JSON، یا غیر تعاون یافتہ محفوظ ورژن خالی سلاٹ نہیں ہیں۔

نسخہ اسناد کی تجدید نہیں کرتا ہے۔ براؤزر SDK لانچ ٹوکن کی تجدید کرتا ہے جب تک وہ باقی رہتے ہیں۔ درست زیر التواء شرائط کو حل کریں یا دوبارہ پڑھنے کی کوشش کرنے سے پہلے گیم کے داخلے کے بہاؤ کے ذریعے دوبارہ تصدیق کریں۔

حالات اور تنازعات

جو بھری ہوئی تھی۔ اگلی تحریر پر ہیڈر اگر کوئی دوسرا آلہ لکھتا ہے۔
ETag کے ساتھ موجودہ بچت If-Match: <etag> 412 precondition_failed
کوئی بچت نہیں (404) If-None-Match: * 412 precondition_failed
کچھ بھی کامیابی سے لوڈ نہیں ہوا۔ نہیں لکھنا پڑھنے کی دوبارہ کوشش کرتے رہیں یا کھلاڑی کو بغیر ٹریک کیے ہوئے کھیل میں رہنے دیں۔

سے ورژن استعمال کریں۔ ڈیٹا جو آپ نے لوڈ کیا۔، بعد میں میٹا ڈیٹا چیک نہیں۔ واپس کیے گئے نئے ETag کو برقرار رکھیں ہر کامیاب تحریر کے ذریعے۔ اے 412 بلاکس مزید بچاتا ہے جب تک کہ گیم موجودہ ورژن کو لوڈ نہیں کرتا ہے۔ کھلاڑی اسے اپناتا ہے یا ضم کرتا ہے۔ خاموشی سے حالت کو نہ ہٹائیں اور کسی دوسرے آلے کو اوور رائٹ نہ کریں۔ API اس اوور رائٹ کو کالعدم کرنے کے لیے کوئی محفوظ تاریخ پیش نہیں کرتا ہے۔

اگر دو آلات کو آزادانہ طور پر کاؤنٹرز یا انوینٹری کو اپ ڈیٹ کرنا ضروری ہے، تو ایک آخری تحریر جیتنے والی دستاویز کھیل فٹ نہیں ہو سکتا. سرور کی مستند حالت کا استعمال کریں اور اس کے بجائے توثیق شدہ گیم کمانڈز کی وضاحت کریں۔

فارمیٹ اور ہجرت کو محفوظ کریں۔

ایک فارمیٹ ورژن شامل کریں اور ریموٹ کو اپنانے سے پہلے اقسام، حدود اور مطلوبہ فیلڈز کی توثیق کریں ترقی معاون پرانے ورژنز کے لیے ایک منتقلی لکھیں۔ ایک نئے نامعلوم ورژن کو برقرار رکھیں؛ اسے ڈیفالٹس پر دوبارہ ترتیب نہ دیں اور اس پر اپ لوڈ کریں۔ کلاؤڈ ڈیٹا کلائنٹ کے زیر کنٹرول ہے، لہذا اس کا علاج کبھی نہ کریں۔ مسابقتی سکور یا ادا شدہ استحقاق کے ثبوت کے طور پر۔

ڈیفالٹ ڈی کوڈ کی بچت کی حد 10 MiB ہے۔ Base64 درخواست کے جسم میں تقریبا ایک تہائی کا اضافہ کرتا ہے۔ 413 cloud_save_too_large, 409 cloud_save_slots_exhausted، اور 507 cloud_save_quota_exceeded مختلف صلاحیت کی ناکامیاں ہیں؛ معائنہ code اور limit. عزت Retry-After پر 429.

چیک پوائنٹس اور صفحہ سے باہر نکلنا

بامعنی چوکیوں پر محفوظ کریں اور دکھائی دینے والی سیو ایکشن پیش کریں۔ تیز رفتار تبدیلیوں کو ڈیباؤنس کریں اور سیریلائز کریں۔ لکھتا ہے لہذا ہر ایک آخری تسلیم شدہ ورژن استعمال کرتا ہے۔ تسلیم ہونے تک گندا اشارے رکھیں۔ ایک ناکام درخواست ایک تصدیق شدہ بچت نہیں ہے۔ ایک مبہم تحریر کے بعد، دوسری تحریر سے پہلے دوبارہ لوڈ کریں۔

صفحہ سے باہر نکلنا اپ لوڈ بہترین کوشش ہے۔ براؤزر fetch(..., { keepalive: true }) ایک چھوٹی سی پرواز میں حصہ لیتا ہے۔ باڈی الاؤنس، عام طور پر 64 KiB؛ ایک 10 MiB بچت اس پر بھروسہ نہیں کر سکتی۔ کسی بھی درخواست سے پہلے ایک ٹیب بند ہو سکتا ہے۔ ختم پائیداری کے لیے عام پیش منظر کی بچت کا استعمال کریں، اور غیر تسلیم شدہ پیش رفت کا لیبل نہ لگائیں۔ محفوظ کر لیا Quickstart ہر پوائنٹ کلیکشن پر اپنی چھوٹی دستاویز کو محفوظ کرتا ہے۔

اپنے گیم میں تصدیق کریں۔

  1. ایک خالی سلاٹ کے ساتھ شروع کریں؛ پیشرفت جمع کریں، محفوظ کریں، دوبارہ لوڈ کریں، اور تصدیق کریں کہ اس کی واپسی ہے۔
  2. نیٹ ورک کی خرابی کے ساتھ پڑھنے سے انکار کریں، 403، یا غلط ڈیٹا؛ تصدیق کریں کہ کوئی اپ لوڈ نہیں ہے۔
  3. ایک ورژن پر دو ٹیبز کھولیں۔ A میں محفوظ کریں، پھر B: B کو تنازعات کو سامنے لانا چاہیے اور اپنی مقامی حالت کو محفوظ رکھنا چاہیے۔
  4. نامعلوم ڈیٹا کو اوور رائٹ کیے بغیر پرانے/نئے فارمیٹ اور گمشدہ اعتراف کی جانچ کریں۔
  5. غیر ٹریک شدہ سنگل پلیئر موڈ میں داخل ہوں اور تصدیق کریں کہ پلے کلاؤڈ سیو کی کوئی درخواست نہیں کرتا ہے۔

دیکھیں چلانے کے قابل کوئیک اسٹارٹ, API کنونشنز، اور خرابیوں کا سراغ لگانا.
